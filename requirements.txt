    <script>
        const fileInput = document.getElementById('fileInput');
        const dropzone = document.getElementById('dropzone');
        const previewContainer = document.getElementById('previewContainer');
        const imagePreview = document.getElementById('imagePreview');
        const fileNameLabel = document.getElementById('fileName');
        const submitBtn = document.getElementById('submitBtn');
        const uploadForm = document.getElementById('uploadForm');

        const emptyState = document.getElementById('emptyResultState');
        const loadingState = document.getElementById('loadingState');
        const resultsPanel = document.getElementById('resultsPanel');

        fileInput.addEventListener('change', e => e.target.files[0] && handleFileSelection(e.target.files[0]));

        function handleFileSelection(file) {
            fileNameLabel.textContent = file.name;
            submitBtn.disabled = false;
            submitBtn.className = "w-full bg-gradient-to-r from-teal-500 to-emerald-500 text-zinc-950 font-bold py-3.5 px-4 rounded-xl hover:opacity-90 active:scale-[0.99] transition-all flex items-center justify-center gap-2 shadow-lg shadow-teal-500/10 cursor-pointer";

            const reader = new FileReader();
            reader.onload = e => {
                imagePreview.src = e.target.result;
                previewContainer.classList.remove('hidden');
            }
            reader.readAsDataURL(file);
        }

        uploadForm.addEventListener('submit', async function(e) {
            e.preventDefault();
            const file = fileInput.files[0];
            if (!file) return;

            const formData = new FormData();
            formData.append('file', file);

            emptyState.classList.add('hidden');
            resultsPanel.classList.add('hidden');
            loadingState.classList.remove('hidden');

            try {
                const response = await fetch('/predict', { method: 'POST', body: formData });
                const data = await response.json();

                loadingState.classList.add('hidden');

                if (data.success) {
                    document.getElementById('topClass').textContent = data.top_prediction.class;
                    document.getElementById('topConfidence').textContent = data.top_prediction.confidence.toFixed(1) + '%';

                    const listContainer = document.getElementById('predictionsList');
                    listContainer.innerHTML = '';

                    data.all_predictions.forEach(pred => {
                        const isTop = pred.class === data.top_prediction.class;
                        const barColor = isTop ? 'bg-gradient-to-r from-teal-500 to-emerald-400' : 'bg-zinc-700';
                        const textStyle = isTop ? 'text-teal-400 font-bold' : 'text-zinc-400';

                        listContainer.insertAdjacentHTML('beforeend', `
                            <div>
                                <div class="flex justify-between text-sm mb-1.5">
                                    <span class="${isTop ? 'text-white font-semibold flex items-center gap-1.5' : 'text-zinc-300'}">
                                        ${isTop ? '<i class="fa-solid fa-caret-right text-teal-400"></i>' : ''}${pred.class}
                                    </span>
                                    <span class="${textStyle}">${pred.confidence.toFixed(1)}%</span>
                                </div>
                                <div class="w-full bg-zinc-900 h-2.5 rounded-full overflow-hidden border border-zinc-800/50">
                                    <div class="${barColor} h-full transition-all duration-500" style="width: ${pred.confidence}%"></div>
                                </div>
                            </div>
                        `);
                    });

                    resultsPanel.classList.remove('hidden');
                } else {
                    alert('Error: ' + data.error);
                    emptyState.classList.remove('hidden');
                }
            } catch (err) {
                loadingState.classList.add('hidden');
                emptyState.classList.remove('hidden');
                alert('Connection Error: ' + err.message);
            }
        });

        // Enhanced Drag & Drop styling
        ['dragenter', 'dragover'].forEach(eventName => {
            dropzone.addEventListener(eventName, (e) => {
                e.preventDefault();
                dropzone.classList.add('border-teal-400', 'bg-teal-500/10', 'shadow-2xl', 'shadow-teal-500/20');
            }, false);
        });
        ['dragleave', 'drop'].forEach(eventName => {
            dropzone.addEventListener(eventName, (e) => {
                e.preventDefault();
                dropzone.classList.remove('border-teal-400', 'bg-teal-500/10', 'shadow-2xl', 'shadow-teal-500/20');
            }, false);
        });
        dropzone.addEventListener('drop', (e) => {
            if(e.dataTransfer.files.length) {
                fileInput.files = e.dataTransfer.files;
                handleFileSelection(e.dataTransfer.files[0]);
            }
        });
    </script>
